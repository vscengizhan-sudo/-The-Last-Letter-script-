-- Tab Navigation
local tabButtons = {}

local function switchTab(tabName)
    currentTab = tabName
    MainTabFrame.Visible = (tabName == "Main")
    DictTabFrame.Visible = (tabName == "Dictionary")
    SettingsTabFrame.Visible = (tabName == "Settings")
    
    for name, btn in pairs(tabButtons) do
        local active = (name == tabName)
        btn.BackgroundColor3 = active and Color3.fromRGB(45, 45, 60) or Color3.fromRGB(28, 28, 36)
        btn.TextColor3 = active and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(150, 150, 170)
    end
end

local function createTabBtn(name, icon, tabKey, active)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, -12, 0, 34)
    btn.Position = UDim2.new(0, 6, 0, 0)
    btn.BackgroundColor3 = active and Color3.fromRGB(45, 45, 60) or Color3.fromRGB(28, 28, 36)
    btn.BorderSizePixel = 0
    btn.Text = "  " .. icon .. "  " .. name
    btn.TextColor3 = active and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(150, 150, 170)
    btn.Font = Enum.Font.GothamSemibold
    btn.TextSize = 13
    btn.TextXAlignment = Enum.TextXAlignment.Left
    btn.Parent = Sidebar
    
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 7)
    corner.Parent = btn
    
    tabButtons[tabKey] = btn
    
    btn.MouseButton1Click:Connect(function()
        switchTab(tabKey)
    end)
    return btn
end

createTabBtn("Main", "📊", "Main", false)
createTabBtn("A-Z Dictionary", "📜", "Dictionary", true)
createTabBtn("Settings", "⚙️", "Settings", false)

-- Update List Function
local function updateList(filterText)
    for _, child in pairs(ScrollFrame:GetChildren()) do
        if child:IsA("TextButton") then
            child:Destroy()
        end
    end
    
    wordButtons = {}
    local filter = (filterText or ""):lower()
    
    for _, word in ipairs(fullDictionary) do
        if filter == "" or word:lower():sub(1, #filter) == filter then
            local WordBtn = Instance.new("TextButton")
            WordBtn.Size = UDim2.new(1, -6, 0, 28)
            WordBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 40)
            WordBtn.BorderSizePixel = 0
            WordBtn.Text = "     " .. word
            WordBtn.TextColor3 = Color3.fromRGB(220, 220, 240)
            WordBtn.TextXAlignment = Enum.TextXAlignment.Left
            WordBtn.TextSize = 13
            WordBtn.Font = Enum.Font.GothamMedium
            WordBtn.Parent = ScrollFrame
            
            local BtnCorner = Instance.new("UICorner")
            BtnCorner.CornerRadius = UDim.new(0, 6)
            BtnCorner.Parent = WordBtn
            
            table.insert(wordButtons, WordBtn)
            
            WordBtn.MouseEnter:Connect(function()
                if not rainbowEnabled then
                    TweenService:Create(WordBtn, TweenInfo.new(0.15), {
                        BackgroundColor3 = Color3.fromRGB(50, 50, 75),
                        TextColor3 = Color3.fromRGB(255, 255, 255)
                    }):Play()
                end
            end)
            
            WordBtn.MouseLeave:Connect(function()
                if not rainbowEnabled then
                    TweenService:Create(WordBtn, TweenInfo.new(0.15), {
                        BackgroundColor3 = Color3.fromRGB(30, 30, 40),
                        TextColor3 = Color3.fromRGB(220, 220, 240)
                    }):Play()
                end
            end)
        end
    end
end

-- Rainbow Toggle Event
RainbowToggleBtn.MouseButton1Click:Connect(function()
    rainbowEnabled = not rainbowEnabled
    if rainbowEnabled then
        RainbowToggleBtn.Text = "  🌈 Rainbow Words Text: ON"
        RainbowToggleBtn.TextColor3 = Color3.fromRGB(100, 255, 150)
    else
        RainbowToggleBtn.Text = "  🌈 Rainbow Words Text: OFF"
        RainbowToggleBtn.TextColor3 = Color3.fromRGB(200, 200, 220)
        for _, btn in ipairs(wordButtons) do
            btn.TextColor3 = Color3.fromRGB(220, 220, 240)
        end
    end
end)

-- Rainbow Rendering
RunService.RenderStepped:Connect(function()
    if rainbowEnabled then
        local hue = (tick() % 3) / 3
        local rainbowColor = Color3.fromHSV(hue, 0.8, 1)
        for _, btn in ipairs(wordButtons) do
            if btn and btn.Parent then
                btn.TextColor3 = rainbowColor
            end
        end
    end
end)

-- Search Box Signal
SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
    updateList(SearchBox.Text)
end)
